cmake_minimum_required(VERSION 3.21)
project(WinDiskImager VERSION 1.7.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

# Svg is not called directly: it is linked so Qt's SVG icon engine is part of
# the build, and so windeployqt and tools/deploy-cross.sh both pull the
# iconengines/qsvgicon plugin the button icons need at runtime.
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)

# Compressed images (.img.gz / .img.xz) are decompressed as they are written,
# so no expanded copy is ever put on disk.
find_package(ZLIB REQUIRED)
find_package(LibLZMA REQUIRED)

# When cross-compiling, Qt6::lrelease resolves to a Windows .exe that cannot run
# on the build host. Point LRELEASE_EXECUTABLE at the host's native lrelease
# instead (e.g. -DLRELEASE_EXECUTABLE=/usr/bin/lrelease-qt6).
if(NOT LRELEASE_EXECUTABLE)
    find_package(Qt6 REQUIRED COMPONENTS LinguistTools)
    set(LRELEASE_EXECUTABLE Qt6::lrelease)
endif()

set(LANGUAGES es it pl nl de fr zh_CN zh_TW ta_IN ko ja)

# translations.qrc references lang/*.qm relative to this dir, so build them in-tree.
set(QM_FILES "")
foreach(lang ${LANGUAGES})
    set(ts "${CMAKE_CURRENT_SOURCE_DIR}/lang/diskimager_${lang}.ts")
    set(qm "${CMAKE_CURRENT_SOURCE_DIR}/lang/diskimager_${lang}.qm")
    add_custom_command(OUTPUT ${qm}
        COMMAND ${LRELEASE_EXECUTABLE} ${ts} -qm ${qm}
        DEPENDS ${ts} VERBATIM)
    list(APPEND QM_FILES ${qm})
endforeach()
add_custom_target(translations DEPENDS ${QM_FILES})

# The .ico the resource compiler needs is rendered from the SVG on every build,
# so the icon is edited in one place and nothing has to be regenerated by hand.
#
# mkicon has to run on the machine doing the build, which when cross-compiling is
# not the machine being built for. So it is configured on its own, without the
# cross toolchain, and picks up the host's native Qt.
set(ICON_SVG ${CMAKE_CURRENT_SOURCE_DIR}/images/WinDiskImager.svg)
# Into the build tree, not the source tree. The rendering is identical
# everywhere -- the six DIB sizes come out byte for byte the same on MSYS2 and
# in the Fedora container -- but the 128 and 256 entries are PNG, and different
# zlib versions compress identical pixels to different bytes. Committing the
# result meant every cross build dirtied the working tree over nothing.
#
# So the .ico is no longer a file in the repository: the SVG is the icon, and
# this is a build artefact like any other.
set(ICON_ICO ${CMAKE_CURRENT_BINARY_DIR}/images/WinDiskImager.ico)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/images)
set(MKICON_BUILD ${CMAKE_BINARY_DIR}/mkicon-host)
if(CMAKE_HOST_WIN32)
    set(MKICON_EXE ${MKICON_BUILD}/mkicon.exe)
else()
    set(MKICON_EXE ${MKICON_BUILD}/mkicon)
endif()

# --unset=CC --unset=CXX is the whole trick. While cmake is working out which
# compiler it has, it puts that compiler in its own environment, and a child
# started from here inherits it. On a cross build that child would then compile
# with the MinGW compiler while believing it is targeting Linux, and go looking
# for Qt somewhere it is not -- so mkicon failed to configure with "could not
# find Qt6", on a machine where a native Qt was plainly installed.
#
# It only bit on the FIRST configure: a reconfigure does not detect the compiler
# again, so the environment is clean and the same command works. That is what
# made it look like it depended on the phase of the moon.
#
# Clearing CMAKE_TOOLCHAIN_FILE is belt and braces: cmake 3.21 and later read
# that name from the environment too.
if(EXISTS ${MKICON_BUILD}/CMakeCache.txt)
    file(READ ${MKICON_BUILD}/CMakeCache.txt MKICON_CACHE)
    if(MKICON_CACHE MATCHES "CMAKE_CXX_COMPILER:[^=]*=[^
]*mingw")
        # Left by a run that inherited the cross compiler; it would keep failing.
        file(REMOVE_RECURSE ${MKICON_BUILD})
    endif()
endif()

execute_process(
    COMMAND ${CMAKE_COMMAND} -E env --unset=CC --unset=CXX
            ${CMAKE_COMMAND} -S ${CMAKE_CURRENT_SOURCE_DIR}/../tools/mkicon
                             -B ${MKICON_BUILD} -G ${CMAKE_GENERATOR}
                             -DCMAKE_BUILD_TYPE=Release
                             -DCMAKE_TOOLCHAIN_FILE=
    RESULT_VARIABLE MKICON_CONFIGURED
    OUTPUT_VARIABLE MKICON_OUTPUT
    ERROR_VARIABLE  MKICON_OUTPUT)
if(NOT MKICON_CONFIGURED EQUAL 0)
    message(FATAL_ERROR
        "cannot configure tools/mkicon, which renders the application icon.
"
        "It is built with the host's own compiler and Qt, not the cross ones, "
        "because it has to run here rather than on the machine being built for. "
        "On Fedora that means gcc-c++, qt6-qtbase-devel and qt6-qtsvg-devel; "
        "MSYS2 UCRT64 already has them.

"
        "cmake said:
${MKICON_OUTPUT}")
endif()

# QT_QPA_PLATFORM: mkicon is a QGuiApplication and the container has no display.
add_custom_command(
    OUTPUT ${ICON_ICO}
    COMMAND ${CMAKE_COMMAND} --build ${MKICON_BUILD}
    COMMAND ${CMAKE_COMMAND} -E env QT_QPA_PLATFORM=offscreen
            ${MKICON_EXE} ${ICON_SVG} ${ICON_ICO}
    DEPENDS ${ICON_SVG}
    COMMENT "Rendering the application icon from images/WinDiskImager.svg"
    VERBATIM)
add_custom_target(appicon DEPENDS ${ICON_ICO})

qt_add_executable(WinDiskImager WIN32
    disk.cpp
    imagesource.cpp
    main.cpp
    mainwindow.cpp
    droppablelineedit.cpp
    devicecombobox.cpp
    elapsedtimer.cpp
    mainwindow.ui
    gui_icons.qrc
    translations.qrc
    WinDiskImager.rc
)
add_dependencies(WinDiskImager translations appicon)
# WinDiskImager.rc names the icon by a relative path, and windres looks for it
# along its include path. The generated one lives here.
target_include_directories(WinDiskImager PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
# windres reads the .ico while compiling the resource script.
set_property(SOURCE WinDiskImager.rc APPEND PROPERTY OBJECT_DEPENDS ${ICON_ICO})

# -DTEST_NO_ADMIN=ON builds an executable that asks for no elevation, so the GUI
# can be started and looked at without a UAC prompt every time. Such a build
# cannot open a device for writing: it is for working on the interface, never
# for imaging or for release.
option(TEST_NO_ADMIN "Build without the requireAdministrator manifest (GUI testing only)" OFF)
if(TEST_NO_ADMIN)
    message(WARNING
        "TEST_NO_ADMIN is on: this build requests no elevation and cannot write "
        "to a device. Do not ship it.")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/WinDiskImager.manifest" _manifest)
    string(REPLACE "requireAdministrator" "asInvoker" _manifest "${_manifest}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WinDiskImager-asinvoker.manifest" "${_manifest}")
    # windres looks for the referenced manifest along its include path, which
    # already has the build directory on it.
    set_property(SOURCE WinDiskImager.rc APPEND PROPERTY COMPILE_DEFINITIONS TEST_NO_ADMIN)
    target_compile_definitions(WinDiskImager PRIVATE TEST_NO_ADMIN)
endif()

target_compile_definitions(WinDiskImager PRIVATE
    VER="${PROJECT_VERSION}"
    # The version is written down here and nowhere else. windres runs the C
    # preprocessor over WinDiskImager.rc, so definitions set on the target reach
    # the resource script like any other source and it never repeats the
    # number. Two more spellings of it, because the resource needs shapes that
    # VER cannot give it:
    #   VER_TUPLE  the comma-separated form VERSIONINFO's FILEVERSION takes.
    #   VER_RAW    the dotted form, unquoted. windres drops the quotes from a
    #              -D value -- VER arrives there as bare 1.5.0 and will not
    #              parse as a string -- so the .rc rebuilds the literal with #.
    VER_TUPLE=${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0
    VER_RAW=${PROJECT_VERSION}
    WINVER=0x0601
    _WIN32_WINNT=0x0601
)

target_link_libraries(WinDiskImager PRIVATE Qt6::Widgets Qt6::Svg ZLIB::ZLIB LibLZMA::LibLZMA)

# This code uses the ANSI Win32 API; Qt6 turns UNICODE on by default.
qt6_disable_unicode_defines(WinDiskImager)

set_target_properties(WinDiskImager PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
)
