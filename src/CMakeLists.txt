cmake_minimum_required(VERSION 3.21)
project(Win32DiskImager VERSION 1.4.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)

# Svg is not called directly: it is linked so Qt's SVG icon engine is part of
# the build, and so windeployqt and tools/deploy-cross.sh both pull the
# iconengines/qsvgicon plugin the button icons need at runtime.
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)

# Compressed images (.img.gz / .img.xz) are decompressed as they are written,
# so no expanded copy is ever put on disk.
find_package(ZLIB REQUIRED)
find_package(LibLZMA REQUIRED)

# When cross-compiling, Qt6::lrelease resolves to a Windows .exe that cannot run
# on the build host. Point LRELEASE_EXECUTABLE at the host's native lrelease
# instead (e.g. -DLRELEASE_EXECUTABLE=/usr/bin/lrelease-qt6).
if(NOT LRELEASE_EXECUTABLE)
    find_package(Qt6 REQUIRED COMPONENTS LinguistTools)
    set(LRELEASE_EXECUTABLE Qt6::lrelease)
endif()

set(LANGUAGES es it pl nl de fr zh_CN zh_TW ta_IN ko ja)

# translations.qrc references lang/*.qm relative to this dir, so build them in-tree.
set(QM_FILES "")
foreach(lang ${LANGUAGES})
    set(ts "${CMAKE_CURRENT_SOURCE_DIR}/lang/diskimager_${lang}.ts")
    set(qm "${CMAKE_CURRENT_SOURCE_DIR}/lang/diskimager_${lang}.qm")
    add_custom_command(OUTPUT ${qm}
        COMMAND ${LRELEASE_EXECUTABLE} ${ts} -qm ${qm}
        DEPENDS ${ts} VERBATIM)
    list(APPEND QM_FILES ${qm})
endforeach()
add_custom_target(translations DEPENDS ${QM_FILES})

# The .ico the resource compiler needs is rendered from the SVG on every build,
# so the icon is edited in one place and nothing has to be regenerated by hand.
#
# mkicon has to run on the machine doing the build, which when cross-compiling is
# not the machine being built for. So it is configured on its own, without the
# cross toolchain, and picks up the host's native Qt.
set(ICON_SVG ${CMAKE_CURRENT_SOURCE_DIR}/images/Win32DiskImager.svg)
set(ICON_ICO ${CMAKE_CURRENT_SOURCE_DIR}/images/Win32DiskImager.ico)
set(MKICON_BUILD ${CMAKE_BINARY_DIR}/mkicon-host)
if(CMAKE_HOST_WIN32)
    set(MKICON_EXE ${MKICON_BUILD}/mkicon.exe)
else()
    set(MKICON_EXE ${MKICON_BUILD}/mkicon)
endif()

execute_process(
    COMMAND ${CMAKE_COMMAND} -S ${CMAKE_CURRENT_SOURCE_DIR}/../tools/mkicon
                             -B ${MKICON_BUILD} -G ${CMAKE_GENERATOR}
                             -DCMAKE_BUILD_TYPE=Release
    RESULT_VARIABLE MKICON_CONFIGURED
    OUTPUT_QUIET)
if(NOT MKICON_CONFIGURED EQUAL 0)
    message(FATAL_ERROR
        "cannot configure tools/mkicon, which renders the application icon. "
        "It needs a native Qt 6 with Gui and Svg: on Fedora qt6-qtbase-devel "
        "and qt6-qtsvg-devel, which tools/build-env.sh installs. MSYS2 UCRT64 "
        "already has them.")
endif()

# QT_QPA_PLATFORM: mkicon is a QGuiApplication and the container has no display.
add_custom_command(
    OUTPUT ${ICON_ICO}
    COMMAND ${CMAKE_COMMAND} --build ${MKICON_BUILD}
    COMMAND ${CMAKE_COMMAND} -E env QT_QPA_PLATFORM=offscreen
            ${MKICON_EXE} ${ICON_SVG} ${ICON_ICO}
    DEPENDS ${ICON_SVG}
    COMMENT "Rendering the application icon from images/Win32DiskImager.svg"
    VERBATIM)
add_custom_target(appicon DEPENDS ${ICON_ICO})

qt_add_executable(Win32DiskImager WIN32
    disk.cpp
    imagesource.cpp
    main.cpp
    mainwindow.cpp
    droppablelineedit.cpp
    elapsedtimer.cpp
    mainwindow.ui
    gui_icons.qrc
    translations.qrc
    DiskImager.rc
)
add_dependencies(Win32DiskImager translations appicon)
# windres reads the .ico while compiling the resource script.
set_property(SOURCE DiskImager.rc APPEND PROPERTY OBJECT_DEPENDS ${ICON_ICO})

# -DTEST_NO_ADMIN=ON builds an executable that asks for no elevation, so the GUI
# can be started and looked at without a UAC prompt every time. Such a build
# cannot open a device for writing: it is for working on the interface, never
# for imaging or for release.
option(TEST_NO_ADMIN "Build without the requireAdministrator manifest (GUI testing only)" OFF)
if(TEST_NO_ADMIN)
    message(WARNING
        "TEST_NO_ADMIN is on: this build requests no elevation and cannot write "
        "to a device. Do not ship it.")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/DiskImager.manifest" _manifest)
    string(REPLACE "requireAdministrator" "asInvoker" _manifest "${_manifest}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/DiskImager-asinvoker.manifest" "${_manifest}")
    # windres looks for the referenced file along its include path.
    set_property(SOURCE DiskImager.rc APPEND PROPERTY COMPILE_DEFINITIONS TEST_NO_ADMIN)
    target_include_directories(Win32DiskImager PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_definitions(Win32DiskImager PRIVATE TEST_NO_ADMIN)
endif()

target_compile_definitions(Win32DiskImager PRIVATE
    VER="${PROJECT_VERSION}"
    WINVER=0x0601
    _WIN32_WINNT=0x0601
)

target_link_libraries(Win32DiskImager PRIVATE Qt6::Widgets Qt6::Svg ZLIB::ZLIB LibLZMA::LibLZMA)

# This code uses the ANSI Win32 API; Qt6 turns UNICODE on by default.
qt6_disable_unicode_defines(Win32DiskImager)

set_target_properties(Win32DiskImager PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
)
